=======================================

Investigation window used for the primary case:
earliest="10/15/2025:12:45:00"
latest="10/15/2025:13:15:00"

NOTE:
- Start broad, discover available sourcetypes/EventCodes, then narrow.
- Field names differ between data sources. Sysmon commonly uses SourceIp/DestinationIp,
  while Security/Suricata/Zeek may use src_ip/dest_ip or id_orig_h/id_resp_h.
- Do not label Type 3 network logons as RDP. RDP is normally Type 10.
- Do not claim sustained C2 beaconing unless repeated callback behavior is demonstrated.

19 - Curated incident timeline (duplicates grouped)
---------------------------------------------------
index="mydfir-soc"
earliest="10/15/2025:12:45:00"
latest="10/15/2025:13:15:00"
(
    "172.16.0.184"
    OR "python.exe"
    OR "157.245.46.190"
    OR "ADDC01.KCD.local"
    OR "PythonUpdate"
)
| eval Activity=case(
    EventCode=4625 AND lower(user)="ryan.adams",
        "Failed logons",
    EventCode=4624 AND Logon_Type=3 AND lower(user)="ryan.adams",
        "Successful network logon",
    EventCode=11 AND like(TargetFilename,"%Ryan.Adams%Music%python.exe"),
        "python.exe created",
    EventCode=1 AND like(Image,"%Ryan.Adams%Music%python.exe"),
        "python.exe executed",
    EventCode=3
        AND like(Image,"%Ryan.Adams%Music%python.exe")
        AND DestinationIp="157.245.46.190"
        AND DestinationPort=8888,
        "External connection",
    EventCode=22
        AND like(Image,"%Ryan.Adams%Music%python.exe")
        AND QueryName="ADDC01.KCD.local",
        "DNS lookup for ADDC01",
    EventCode=3
        AND like(Image,"%Ryan.Adams%Music%python.exe")
        AND DestinationIp="172.16.0.7",
        "Internal RPC-related connection",
    EventCode=1
        AND like(CommandLine,"%PythonUpdate%")
        AND like(CommandLine,"%schtasks.exe%"),
        "Scheduled task persistence"
)
| where isnotnull(Activity)
| eval Source=case(
    Activity="Failed logons", src_ip,
    Activity="Successful network logon", src_ip,
    true(), SourceIp
)
| eval Destination=if(
    isnotnull(DestinationIp),
    DestinationIp . if(isnotnull(DestinationPort), ":" . DestinationPort, ""),
    null()
)
| eval Detail=case(
    Activity="Failed logons",
        "Source workstation: " . Workstation_Name,
    Activity="Successful network logon",
        "Source workstation: " . Workstation_Name . " | NTLM Type 3",
    Activity="python.exe created",
        TargetFilename,
    Activity="python.exe executed",
        Image,
    Activity="External connection",
        Image,
    Activity="DNS lookup for ADDC01",
        QueryName . " -> " . QueryResults,
    Activity="Internal RPC-related connection",
        Image,
    Activity="Scheduled task persistence",
        CommandLine
)
| stats
    earliest(_time) as _time
    count as Count
    values(user) as User
    values(Source) as Source
    values(Destination) as Destination
    values(Detail) as Detail
    by Activity
| sort _time
| eval Time=strftime(_time,"%Y-%m-%d %H:%M:%S")
| table Time Activity Count User Source Destination Detail
